class Solution {
    public int minCostToMoveChips(int[] nums) {
        int oddCount = 0;
        int evenCount = 0;

        for (int num : nums) {
            if(num % 2 == 0){
                evenCount++;
            }else{
                oddCount++;
            }
        }
        return Math.min(oddCount,evenCount);
    }
}
