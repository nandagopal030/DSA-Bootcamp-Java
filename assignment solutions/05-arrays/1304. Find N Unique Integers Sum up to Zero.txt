1304. Find N Unique Integers Sum up to Zero

class Solution {
    public int[] sumZero(int n) {
        int[] arr = new int[n];
        int num = 1;
        for (int i = 0; i < n / 2; i++) {
            arr[i] = num;
            arr[n - 1 - i ] = -num;
            num++;
        }
        return arr;
    }
}
