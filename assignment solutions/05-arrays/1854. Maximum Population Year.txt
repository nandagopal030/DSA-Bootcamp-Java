class Solution {
    public int maximumPopulation(int[][] logs) {
        int[] population = new int[102];

        for(int[] log : logs){
            population[log[0] - 1950]++;
            population[log[1] - 1950]--;
        }
        int maxYear = 1950;
        int maxPop = 0;
        int currPop = 0;

        for(int i =0; i < 102; i++){
            currPop += population[i];
            if(currPop > maxPop){
                maxPop = currPop;
                maxYear = 1950+ i;
            }
        }
        return maxYear;
    }
}
