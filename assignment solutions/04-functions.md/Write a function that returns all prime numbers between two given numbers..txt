import java.util.*;
//Write a function that returns all prime numbers between two given numbers.
public class Main
{
	public static void main(String[] args) {
	    int n = 1;
	    int m = 10;
	    isPrimeRange(n, m);
	}
	static boolean isPrime(int n){
	    if(n <= 1){
	        return false;
	    }
	    for(int i = 2; i* i <= n; i++){
	        if(n % i == 0){
	            return false;
	        }
	    }
	    return true;
	}
	
	static int isPrimeRange(int n, int m){
	   int[] temp = new int[m - n  + 1];
	   int index = 0;
	    for(int i = n; i<=m; i++){
	        if(isPrime(i)){
	            temp[index++] = i;
	        }
	    }   
	    System.out.println(Arrays.toString(temp));
	    return  1;
	}
	
}
