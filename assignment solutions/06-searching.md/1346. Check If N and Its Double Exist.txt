class Solution {
    public boolean checkIfExist(int[] arr) {
        Arrays.sort(arr);

        for (int i = 0; i < arr.length - 1; i++) {
            int target = 2 * arr[i];
            if (binarySearch(arr, target, i)) {
                return true;
            }
        }
        return false;
    }

    static boolean binarySearch(int[] arr, int target, int index) {
        int start = 0;
        int end = arr.length - 1;

        while (start <= end) {
            int mid = start + (end - start) / 2;
           
            if (arr[mid] == target) {
                if (mid != index) {
                    return true;
                }else{
                    start = mid + 1;
                }
            } else if (arr[mid] < target) {
                start = mid + 1;
            } else {
                end = mid - 1;
            }
        }
        return false;
    }
}aa
