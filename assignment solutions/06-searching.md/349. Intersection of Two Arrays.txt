class Solution {
    public int[] intersection(int[] nums1, int[] nums2) {
        HashSet<Integer> set = new HashSet<>();
        Arrays.sort(nums1);
        Arrays.sort(nums2);

        for (int num : nums1) {
            if (binarySearch(num, nums2)) {
                set.add(num);
            }
        }
        int[] arr = new int[set.size()];
        int i = 0;
        for (int num : set) {
            arr[i] = num;
            i++;
        }
        return arr;

    }

    static boolean binarySearch(int target, int[] nums2) {
        int start = 0;
        int end = nums2.length - 1;
        while (start <= end) {
            int mid = start + (end - start) / 2;
            if (nums2[mid] == target)
                return true;
            else if (nums2[mid] < target)
                start = mid + 1;
            else
                end = mid - 1;
        }
        return false;
    }

}
