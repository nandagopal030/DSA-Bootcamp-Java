import java.util.*;
// Calculate Distance Between Two Points
public class Main
{
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);

		System.out.println("Enter the points x1 and x2");
		int x1  = sc.nextInt();
		int x2 = sc.nextInt();
		
		System.out.println("Enter the points y1 and y2");
		int y1 = sc.nextInt();
		int y2 = sc.nextInt();
		
		double overAllX = Math.pow((x2 - x1) , 2);
		double overAllY = Math.pow((y2 - y1),  2);
		double d = overAllY + overAllX;
		double result = Math.sqrt(d);
		System.out.println(result);
	}
}
